30void dfs(int u) {
31 stav[u] = 1; // sivy — prave navstevujeme u (je na zasobniku rekurzie)
32
33 for (int v : adj[u]) {
34 if (stav[v] == 1) {
35 // ak je sused sivy tak mame cyklus
36 cyklus = true;
37 return;
38 }
39 if (stav[v] == 0) {
40 dfs(v); // rekurzia
41 }
42
43 if (cyklus) return; // ak je cyklus tak koncime rekurziu
44 }
45
46 stav[u] = 2; // cierny — vsetci susedia spracovani
47 topo.push_back(u); // pridame do topologickeho poradia (odzadu)
48}

50bool jeDAG() {
51 stav.assign(n, 0); // vsetky vrcholy su biele, pripravene
52 cyklus = false;
53 topo.clear();
54
55 // spustime DFS z kazdeho vrcholu
56 for (int i = 0; i < n; i++) {
57 if (stav[i] == 0) {
58 dfs(i);
59 }
60 }
61
62 // DFS pridava odzadu preto my potrebujeme topo otocit
63 reverse(topo.begin(), topo.end());
64
65 return !cyklus;
66}
